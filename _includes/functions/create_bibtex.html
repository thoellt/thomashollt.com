{%- comment -%} Builds a BibTeX entry as plain text in `bibtex` (newline separated, for a downloadable .bib file). Usage: {% include functions/create_bibtex.html publication=publication %} Without a `publication` parameter it uses the current page. {%- endcomment -%} {%- assign pub = include.publication | default: page -%} {%- capture nl %} {% endcapture -%} {%- assign bibtex = "" -%} {%- comment %} type {% endcomment -%} {%- if pub.type == "article" or pub.type == "preprint" -%} {%- assign bibtex = bibtex | append: "@article" -%} {%- elsif pub.type == "inproceedings" -%} {%- assign bibtex = bibtex | append: "@inproceedings" -%} {%- elsif pub.type == "inbook" -%} {%- assign bibtex = bibtex | append: "@inbook" -%} {%- elsif pub.type == "book" -%} {%- assign bibtex = bibtex | append: "@book" -%} {%- elsif pub.type == "mscthesis" -%} {%- assign bibtex = bibtex | append: "@mastersthesis" -%} {%- elsif pub.type == "phdthesis" -%} {%- assign bibtex = bibtex | append: "@phdthesis" -%} {%- else -%} {%- assign bibtex = bibtex | append: "@misc" -%} {%- endif -%} {%- comment %} key {% endcomment -%} {%- assign bibtex = bibtex | append: "{" | append: pub.key | append: "," | append: nl -%} {%- comment %} authors, used as written in the publication file {% endcomment -%} {%- assign bibtex = bibtex | append: " author = { " -%} {%- for author in pub.authors -%} {%- include functions/replace_umlauts.html input=author -%} {%- assign bibtex = bibtex | append: fixedString -%} {%- unless forloop.last -%} {%- assign bibtex = bibtex | append: " and " -%} {%- endunless -%} {%- endfor -%} {%- assign bibtex = bibtex | append: " }," | append: nl -%} {%- comment %} title {% endcomment -%} {%- assign bibtex = bibtex | append: " title = { " | append: pub.title | append: " }," | append: nl -%} {%- comment %} journal/proceedings {% endcomment -%} {%- if pub.type == "article" or pub.type == "preprint" -%} {%- assign bibtex = bibtex | append: " journal = { " | append: pub.journal | append: " }," | append: nl -%} {%- elsif pub.type == "inproceedings" -%} {%- assign bibtex = bibtex | append: " booktitle = { " | append: pub.journal | append: " }," | append: nl -%} {%- elsif pub.type == "inbook" -%} {%- assign bibtex = bibtex | append: " booktitle = { " | append: pub.journal | append: " }," | append: nl -%} {%- elsif pub.type == "poster" -%} {%- assign bibtex = bibtex | append: " howpublished = { Poster presentation at " | append: pub.journal | append: " }," | append: nl -%} {%- endif -%} {%- comment %} volume {% endcomment -%} {%- if pub.volume -%} {%- assign bibtex = bibtex | append: " volume = { " | append: pub.volume | append: " }," | append: nl -%} {%- endif -%} {%- comment %} issue {% endcomment -%} {%- if pub.issue -%} {%- assign bibtex = bibtex | append: " number = { " | append: pub.issue | append: " }," | append: nl -%} {%- endif -%} {%- comment %} chapter {% endcomment -%} {%- if pub.chapter -%} {%- assign bibtex = bibtex | append: " chapter = { " | append: pub.chapter | append: " }," | append: nl -%} {%- endif -%} {%- comment %} pages {% endcomment -%} {%- if pub.page_start -%} {%- assign bibtex = bibtex | append: " pages = { " | append: pub.page_start -%} {%- if pub.page_end -%} {%- assign bibtex = bibtex | append: " -- " | append: pub.page_end -%} {%- endif -%} {%- assign bibtex = bibtex | append: " }," | append: nl -%} {%- endif -%} {%- comment %} editor {% endcomment -%} {%- if pub.editor -%} {%- assign bibtex = bibtex | append: " editor = { " | append: pub.editor | append: " }," | append: nl -%} {%- endif -%} {%- comment %} publisher {% endcomment -%} {%- if pub.publisher -%} {%- assign bibtex = bibtex | append: " publisher = { " | append: pub.publisher | append: " }," | append: nl -%} {%- endif -%} {%- comment %} school {% endcomment -%} {%- if pub.school -%} {%- include functions/replace_umlauts.html input=pub.school -%} {%- assign bibtex = bibtex | append: " school = { " | append: fixedString | append: " }," | append: nl -%} {%- endif -%} {%- comment %} year {% endcomment -%} {%- if pub.year -%} {%- assign bibtex = bibtex | append: " year = { " | append: pub.year | append: " }," | append: nl -%} {%- endif -%} {%- comment %} doi {% endcomment -%} {%- if pub.doi -%} {%- assign bibtex = bibtex | append: " doi = { " | append: pub.doi | append: " }," | append: nl -%} {%- endif -%} {%- comment %} close {% endcomment -%} {%- assign bibtex = bibtex | append: "}" -%}
